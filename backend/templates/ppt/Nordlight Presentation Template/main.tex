% ---------- Palette ----------
\definecolor{bgdark}{HTML}{0F1B2D}
\definecolor{panel}{HTML}{16273F}
\definecolor{accent}{HTML}{4FD1C5}
\definecolor{accent2}{HTML}{F4A259}
\definecolor{txtmain}{HTML}{ECF0F3}
\definecolor{txtdim}{HTML}{9FB0C3}

\setbeamercolor{background canvas}{bg=bgdark}
\setbeamercolor{normal text}{fg=txtmain,bg=bgdark}
\setbeamercolor{frametitle}{fg=txtmain}
\setbeamercolor{title}{fg=txtmain}
\setbeamercolor{itemize item}{fg=accent}
\setbeamercolor{itemize subitem}{fg=accent2}
\setbeamercolor{block title}{fg=bgdark,bg=accent}
\setbeamercolor{block body}{fg=txtmain,bg=panel}
\setbeamercolor{alerted text}{fg=accent2}
\setbeamercolor{page number in head/foot}{fg=txtdim}

\setbeamerfont{frametitle}{series=\bfseries,size=\Large}
\setbeamerfont{title}{series=\bfseries,size=\huge}

\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{itemize item}{\faCircle}
\setbeamertemplate{itemize subitem}{\faAngleRight}

% ---------- Custom frame title with accent underline ----------
\setbeamertemplate{frametitle}{
  \vspace{4mm}
  \hspace{6mm}\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle\\[1mm]
  \hspace{6mm}\begin{tikzpicture}
    \draw[accent,line width=1.4pt] (0,0) -- (2.1,0);
  \end{tikzpicture}
}

% ---------- Footline: slide number + thin accent line ----------
\setbeamertemplate{footline}{
  \begin{tikzpicture}[remember picture,overlay]
    \draw[panel,line width=0.6pt] ($(current page.south west)+(0,0.62)$) -- ($(current page.south east)+(0,0.62)$);
    \node[anchor=south east,txtdim,font=\scriptsize] at ($(current page.south east)+(-0.35,0.25)$) {\insertframenumber\,/\,\inserttotalframenumber};
    \node[anchor=south west,accent,font=\scriptsize\bfseries] at ($(current page.south west)+(0.35,0.25)$) {NORDLIGHT};
  \end{tikzpicture}
}

\begin{frame}{Problem Statement}
\begin{block}{Motivation}
Default templates often look generic; a distinctive visual identity improves audience engagement.
\end{block}
\begin{block}{Objective}
Design a clean, high-contrast deck that still compiles anywhere, with no external fonts required.
\end{block}
\end{frame}

\begin{frame}{Methodology}
\begin{enumerate}
  \item Define a small, purposeful color palette.
  \item Rebuild title page, headers, and footer with TikZ.
  \item Keep body content simple: itemize, blocks, tables, equations.
  \item Compile and verify with plain \texttt{pdflatex}.
\end{enumerate}
\end{frame}
