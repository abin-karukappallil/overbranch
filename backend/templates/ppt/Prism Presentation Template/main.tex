\begin{frame}{Problem Statement}
\begin{block}{Motivation}
Energetic color and shape can make technical talks feel more approachable.
\end{block}
\begin{block}{Objective}
Deliver a bold, memorable look while keeping the content structure simple and academic.
\end{block}
\end{frame}

\begin{frame}{Methodology}
\begin{enumerate}
  \item Pick 3 vivid accent colors plus a neutral ink tone.
  \item Build a distinctive side-bar and corner-tab title style.
  \item Keep body layout simple: bullets, blocks, tables.
  \item Verify the deck compiles cleanly with pdflatex.
\end{enumerate}
\end{frame}
