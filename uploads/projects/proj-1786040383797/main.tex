% Introduction
\section{Introduction}
\begin{frame}{Introduction}
  \begin{block}{What is KSRTC?}
    The Kerala State Road Transport Corporation (KSRTC) is the state-owned road transport corporation of Kerala, India, providing bus services across the state and to neighboring regions.
  \end{block}
  \begin{itemize}
    \item Established in 1965
    \item Operates over 6,000 buses
    \item Serves more than 100 million passengers annually
  \end{itemize}
\end{frame}
